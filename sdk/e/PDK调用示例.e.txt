.程序集 窗口程序集_启动窗口

.子程序 _按钮_开始_被单击
    .局部变量 h, 整数型
    .局部变量 返回JSON, 文本型
    .局部变量 解析, 类_json
    .局部变量 code, 整数型
    .局部变量 enc, 文本型
    .局部变量 plain, 文本型
    .局部变量 trace, 文本型
    .局部变量 手机, 文本型
    .局部变量 密码, 文本型

    手机 ＝ "13800138000"
    密码 ＝ "Pdk12345678"

    ' —— 1) 创建实例（base_url 留空用默认；生产改成你的 https 域名）——
    h ＝ pdk_create ("http://localhost:8080", "", "")
    .如果 (h ＝ 0)
        信息框 ("创建 PDK 实例失败", 0, , )
        返回 ()
    .如果结束

    ' —— 1.5) 启用协议信封加密（防抓包：所有 body 自动 RSA+AES 加密）——
    '   P0 公钥指纹钉扎（防 MITM 替换公钥，二选一）：
    '   方案 A（生产推荐）：显式预置期望指纹（通过独立可信渠道获取，这里示例用占位串），
    '       之后 pdk_refresh_crypto_config 拉到的公钥会强制比对指纹，不符拒绝启用。
    pdk_set_public_key_pin (h, "a1b2c3d4e5f6...")  ' 替换为真实指纹（32 位 hex）
    '   方案 B（便捷/弱）：不预置指纹时，设置本地钉扎存储路径，首次成功会信任并写入（TOFU），
    '       后续拉取比对；仅防后续替换，首次仍可能被骗，生产不建议单独使用。
    '   pdk_set_pin_store_path (h, "C:\pdk_client\pin.json")
    '   然后一行即可：拉取服务端公钥并按 encryptionMode 自动启用；mode=off 时保持明文（灰度）。
    '   启用后 pdk_login / pdk_acquire_token 等带 body 的请求会自动加密，响应自动解密，
    '   业务代码完全不用改。可选：用 pdk_is_envelope_enabled(h) 确认是否已启用。
    返回JSON ＝ pdk_refresh_crypto_config (h)
    解析.解析 (返回JSON)
    .如果 (到整数 (解析.取属性 ("code").取数值 ()) ＝ 200)
        .如果 (pdk_is_envelope_enabled (h) ＝ 1)
            标签_状态.标题 ＝ "协议加密已启用（kid=" ＋ 解析.取属性 ("data.kid").取文本 () ＋ "）"
        .否则
            标签_状态.标题 ＝ "协议加密未启用（服务端 mode=off，明文传输）"
        .如果结束
    .如果结束

    ' —— 2) 发送短信验证码 ——
    返回JSON ＝ pdk_send_sms (h, 手机, "REGISTER")
    解析.解析 (返回JSON)
    code ＝ 到整数 (解析.取属性 ("code").取数值 ())
    .如果 (code ≠ 200)
        信息框 ("发送短信失败：" ＋ 解析.取属性 ("message").取文本 (), 0, , )
        pdk_destroy (h)
        返回 ()
    .如果结束
    信息框 ("验证码已发送，请查收短信", 0, , )

    ' —— 3) 注册（若已领过体验返回 40010，则改用 pdk_login）——
    返回JSON ＝ pdk_register (h, 手机, 密码, "123456", "")
    解析.解析 (返回JSON)
    code ＝ 到整数 (解析.取属性 ("code").取数值 ())
    .如果 (code ＝ 40010)
        ' 已注册过，直接登录
        返回JSON ＝ pdk_login (h, 手机, 密码)
        解析.解析 (返回JSON)
        code ＝ 到整数 (解析.取属性 ("code").取数值 ())
    .如果结束

    .如果 (code ≠ 200)
        信息框 ("登录/注册失败：" ＋ 解析.取属性 ("message").取文本 (), 0, , )
        pdk_destroy (h)
        返回 ()
    .如果结束

    ' —— 4) 轮询状态，刷新界面（状态枚举见 README 表格）——
    .如果 (pdk_get_last_state (h) ＝ 6)   ' 6 = LoggedIn
        标签_状态.标题 ＝ "已登录：" ＋ pdk_get_phone (h)
        标签_状态.文本颜色 ＝ #绿色
    .如果结束

    ' —— 5) 申请短效加密 Token ——
    返回JSON ＝ pdk_acquire_token (h, "GOODS_COLLECT", "881920391204")
    解析.解析 (返回JSON)
    .如果 (到整数 (解析.取属性 ("code").取数值 ()) ＝ 200)

        ' 取 data.encryptedPayload 并解密
        enc ＝ 解析.取属性 ("data.encryptedPayload").取文本 ()
        plain ＝ pdk_decrypt_token (h, enc)
        .如果 (plain ＝ "" 或 plain ＝ "NULL")
            信息框 ("Token 解密失败（时间窗过期或数据损坏）", 0, , )
        .否则
            信息框 ("解密成功，明文 Token 报文：" ＋ #换行符 ＋ plain, 0, , )
            ' ★ 这里用 plain 里的拼多多 Session 向官方发起采集请求 ★

            ' 取租约号，上报结果（SUCCESS 扣 1 次）
            trace ＝ 解析.取属性 ("data.leaseTraceId").取文本 ()
            返回JSON ＝ pdk_report_result (h, trace, "SUCCESS", 1200, "")
            解析.解析 (返回JSON)
            信息框 ("上报结果：" ＋ 解析.取属性 ("message").取文本 (), 0, , )
        .如果结束
    .否则
        ' 40103=设备互踢 / 40302=配额耗尽 / 50301=槽位繁忙
        信息框 ("申请 Token 失败：" ＋ 解析.取属性 ("message").取文本 (), 0, , )
        .如果 (到整数 (解析.取属性 ("code").取数值 ()) ＝ 40103)
            信息框 ("账号已在其他电脑登录，请先在原设备解绑", 0, , )
        .如果结束
    .如果结束

    ' —— 6) 查询配额（示例）——
    返回JSON ＝ pdk_profile (h)
    解析.解析 (返回JSON)
    信息框 ("剩余次数：" ＋ 解析.取属性 ("data.remainingCalls").取文本 (), 0, , )

    ' —— 7) 释放 ——
    pdk_destroy (h)
